----------------------------------------------------------------
  6. HOW TO USE THE MODEL
----------------------------------------------------------------

  REQUIREMENTS:
    pip install lightgbm joblib pandas scikit-learn


  --- A. Load and predict on new data (from a CSV file) ---

    import pandas as pd
    import joblib

    # Load model
    model = joblib.load("lightgbm_model.pkl")

    # Load your new data (must have same 49 features, no label column)
    new_data = pd.read_csv("your_new_data.csv")

    # Predict class (0 = Legit, 1 = Phishing)
    predictions = model.predict(new_data)

    # Predict probability scores (column 1 = phishing probability)
    probabilities = model.predict_proba(new_data)[:, 1]

    print(predictions)      # e.g. [0, 1, 0, 1, 0]
    print(probabilities)    # e.g. [0.02, 0.98, 0.05, 0.97, 0.01]


  --- B. Predict on a single URL sample ---

    import pandas as pd
    import joblib

    model = joblib.load("lightgbm_model.pkl")

    # Build a single-row DataFrame with all 49 feature values
    # Use -1 for any feature that could not be extracted
    sample = pd.DataFrame([{
        "url_has_ip"               : 0,
        "url_length"               : 75,
        "url_length_flag"          : 1,
        "uses_url_shortener"       : 0,
        "url_has_at_symbol"        : 0,
        "url_double_slash_redirect": 0,
        "url_prefix_suffix_dash"   : 1,
        "url_subdomain_count"      : 3,
        "url_subdomain_flag"       : 1,
        "url_special_char_count"   : 5,
        "url_has_suspicious_kw"    : 1,
        "url_dot_count"            : 4,
        # ... fill remaining 37 features the same way ...
    }])

    pred  = model.predict(sample)[0]
    prob  = model.predict_proba(sample)[0][1]

    label = "PHISHING" if pred == 1 else "LEGIT"
    print(f"Prediction : {label}")
    print(f"Confidence : {prob:.2%}")


  --- C. Evaluate against a labelled test set ---

    import pandas as pd
    import joblib
    from sklearn.metrics import classification_report, roc_auc_score

    model   = joblib.load("lightgbm_model.pkl")
    test_df = pd.read_csv("test_data.csv")

    X_test  = test_df.drop(columns=["label"])
    y_test  = test_df["label"]

    y_pred  = model.predict(X_test)
    y_prob  = model.predict_proba(X_test)[:, 1]

    print(classification_report(y_test, y_pred,
                                target_names=["Legit", "Phishing"]))
    print(f"ROC-AUC: {roc_auc_score(y_test, y_prob):.4f}")


  --- D. Get feature importance ---

    import joblib
    import pandas as pd

    model = joblib.load("lightgbm_model.pkl")

    feature_names = model.feature_name_
    importances   = model.feature_importances_

    importance_df = pd.DataFrame({
        "feature"   : feature_names,
        "importance": importances
    }).sort_values("importance", ascending=False)

    print(importance_df.head(10))
